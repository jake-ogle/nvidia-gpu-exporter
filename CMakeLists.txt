cmake_minimum_required(VERSION 3.20)
project(nvidia_gpu_exporter VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
set(ENABLE_PUSH OFF CACHE BOOL "" FORCE)
set(ENABLE_COMPRESSION OFF CACHE BOOL "" FORCE)
set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
FetchContent_Declare(prometheus GIT_REPOSITORY https://github.com/jupp0r/prometheus-cpp.git GIT_TAG v1.3.0 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(prometheus)
find_path(NVML_INCLUDE_DIR nvml.h HINTS /usr/local/cuda/include /usr/include/nvidia/gdk)
if(NOT NVML_INCLUDE_DIR)
  message(FATAL_ERROR "Install NVIDIA NVML development headers (nvml.h)")
endif()
add_executable(nvidia-gpu-exporter src/main.cpp)
target_include_directories(nvidia-gpu-exporter PRIVATE ${NVML_INCLUDE_DIR})
target_link_libraries(nvidia-gpu-exporter PRIVATE prometheus-cpp::pull ${CMAKE_DL_LIBS})
target_compile_options(nvidia-gpu-exporter PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS nvidia-gpu-exporter RUNTIME DESTINATION bin)
